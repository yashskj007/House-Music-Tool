FIX 1 — SSE KEEPALIVE
In every route that uses SSE (/api/analyse and /api/recommend), start a keepalive interval immediately after setting SSE headers:

const keepalive = setInterval(() => {
  try { res.write(': keepalive\n\n'); } catch(e) { clearInterval(keepalive); }

FIX 6 — INDEX.HTML SSE RECONNECT
In index.html, wrap the EventSource in a reconnect handler:

let retryCount = 0;
function startSSE(url, onMessage, onDone) {
  const es = new EventSource(url);
  es.onmessage = (e) => {
    const data = JSON.parse(e.data);
    onMessage(data);
    if (data.done) { es.close(); onDone(data); }
  };
  es.onerror = () => {
    es.close();
    if (retryCount < 2) {
      retryCount++;
      setTimeout(() => startSSE(url, onMessage, onDone), 2000);
    } else {
      onDone({ error: 'Connection lost. Please try again.' });
    }
  };
}
