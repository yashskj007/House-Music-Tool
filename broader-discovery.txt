5. DEDUPLICATION (CRITICAL)
After merging all sources, deduplicate by normalized artist+title:

function dedupeCandidates(candidates) {
  const seen = new Map();
  candidates.forEach(c => {
    const key = `${c.artist.toLowerCase().trim()}|${c.title.toLowerCase().trim()}`;
    if (!seen.has(key)) {
      seen.set(key, c);
    } else {
      // Merge sources — track that this candidate appeared in multiple sources
      const existing = seen.get(key);
      if (!existing.also_in) existing.also_in = [];
      existing.also_in.push(c.source);
    }
  });
  return Array.from(seen.values());
}

Candidates appearing in multiple sources get a confidence boost in scoring (cross-source validation).
